% ---------- Experience ----------
\section{Experience}
\entry{PhD Research Scholar}{Department of Bioinformatics, University of North Bengal}{Siliguri, India}{2022-Present}
\begin{itemize}
	\item Designed \textbf{AttnSeq-PPI}, a transfer-learning hybrid attention architecture for sequence-based protein-protein interaction prediction; published in \textit{BBA - Proteins and Proteomics} and released as a public web server.
	\item Built \textbf{ARACoFusion}, an uncertainty-aware calibrated ensemble for PPI network prediction in \textit{Arabidopsis thaliana}, optimising for reliable confidence estimates rather than raw accuracy alone.
	\item Developed \textbf{MoE-Bind}, a sparse mixture-of-experts autoregressive generative model for de novo protein binder design.
	\item Pretrained \textbf{PlantPLM}, a family of domain-adapted ESM-2 protein language models (8M/35M/150M), on 1.11B amino-acid tokens from 4.37M MMseqs2-deduplicated Viridiplantae sequences, reaching 8.98 validation perplexity. Trained end-to-end on a single 12\,GB consumer GPU using mixed precision, gradient checkpointing, and gradient accumulation.
	\item Shipped and maintain three model-backed web platforms (Docker + FastAPI/Flask + Nginx) serving inference and interaction-network visualisation to external users.
\end{itemize}
